#Cambiar la a buscar, por ejemplo translation, en la dirección den curl. Al final viene un archivo después de -o que es un XML, es donde se guarda la informacion

#Ingles - translation
curl "https://ndlsearch.ndl.go.jp/api/sru?operation=searchRetrieve&version=1.2&query=title%3D%22translation%22%20AND%20from%3D1850%20AND%20until%3D1920&maximumRecords=200&recordSchema=dcndl&recordPacking=string&startRecord=1" -o translation_en_1850_1920.xml --silent --show-error --write-out "\nHTTP status: %{http_code}\n"
